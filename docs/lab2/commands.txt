# Lab 2 commands (Windows PowerShell)

Set-Location D:\Prog\fastapi-template

# 1) Create local env file (never commit secrets)
Copy-Item .env.example .env

# 2) Start dev stack (web + postgres), with reload + bind-mount
docker compose -p lab2 -f docker-compose.yml -f docker-compose.dev.yml up -d --build

# 3) Verify containers
docker compose -p lab2 -f docker-compose.yml -f docker-compose.dev.yml ps

# 4) Show installed libraries inside the web container (pip list equivalent)
docker compose -p lab2 -f docker-compose.yml -f docker-compose.dev.yml exec -T web poetry show --no-ansi

# 5) Stop
docker compose -p lab2 -f docker-compose.yml -f docker-compose.dev.yml down
